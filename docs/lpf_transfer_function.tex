\section{S Domain}

\begin{align}
H_{model}(s) &= \frac{1.1592 \times 10^7}{s^2 + 6.6172 \times 10^3 \cdot s + 1.1592 \times 10^7} \\[10pt]
H_{lpf}(s) &= \frac{6.2832 \times 10^4}{s + 6.2832 \times 10^4} \\[10pt]
H(s) &= H_{model}(s) \cdot H_{lpf}(s) \notag \\[5pt]
&= \frac{7.2835 \times 10^{11}}{s^3 + 6.9449 \times 10^4 \cdot s^2 + 4.2736 \times 10^8 \cdot s + 7.2835 \times 10^{11}}
\end{align}
